Xoay Quanh Trục Tự Do (Rodrigues' Rotation)

Xoay một góc $\theta$ quanh vector chuẩn hóa bất kỳ $\mathbf{u} = (u_x, u_y, u_z)$:

$$\mathbf{R}(\mathbf{u}, \theta) = \mathbf{I} + (\sin\theta)[\mathbf{u}]_\times + (1-\cos\theta)[\mathbf{u}]_\times^2$$

Trong đó $[\mathbf{u}]_\times$ là ma trận tích có hướng (skew-symmetric matrix).

Bộ Code Python NumPy: Biến Đổi Ma Trận 4×4 Đầy Đủ 

import numpy as np

class Affine4x4:
    @staticmethod
    def identity():
        return np.eye(4, dtype=np.float64)

    @staticmethod
    def shear(sh_xy=0.0, sh_xz=0.0, sh_yx=0.0, sh_yz=0.0, sh_zx=0.0, sh_zy=0.0):
        """Ma trận biến dạng cắt 4x4 (Shearing)"""
        M = np.eye(4, dtype=np.float64)
        M[0, 1] = sh_xy
        M[0, 2] = sh_xz
        M[1, 0] = sh_yx
        M[1, 2] = sh_yz
        M[2, 0] = sh_zx
        M[2, 1] = sh_zy
        return M

    @staticmethod
    def rodrigues(axis, angle_deg):
        """Công thức xoay Rodrigues quanh trục tự do u = [ux, uy, uz]"""
        u = np.array(axis, dtype=np.float64)
        norm = np.linalg.norm(u)
        if norm < 1e-9:
            raise ValueError("Vector trục không được là vector 0.")
        u = u / norm
        
        theta = np.radians(angle_deg)
        c = np.cos(theta)
        s = np.sin(theta)
        
        # Ma trận phản đối xứng [u]_x
        K = np.array([
            [0.0, -u[2], u[1]],
            [u[2], 0.0, -u[0]],
            [-u[1], u[0], 0.0]
        ])
        
        # R = I + sin(theta)*K + (1 - cos(theta))*(K @ K)
        R3 = np.eye(3) + s * K + (1.0 - c) * (K @ K)
        
        M = np.eye(4)
        M[:3, :3] = R3
        return M

    @staticmethod
    def plane_to_world(origin, normal):
        """Tạo ma trận chuyển đổi từ hệ toạ độ phẳng (Plane) sang World"""
        P0 = np.array(origin, dtype=np.float64)
        W = np.array(normal, dtype=np.float64)
        W = W / np.linalg.norm(W)
        
        # Xây dựng hệ cơ sở trực chuẩn U, V, W
        ref = np.array([0.0, 1.0, 0.0]) if abs(W[1]) < 0.9 else np.array([1.0, 0.0, 0.0])
        U = np.cross(ref, W)
        U = U / np.linalg.norm(U)
        V = np.cross(W, U)
        V = V / np.linalg.norm(V)
        
        M = np.eye(4)
        M[:3, 0] = U
        M[:3, 1] = V
        M[:3, 2] = W
        M[:3, 3] = P0
        return M

    @staticmethod
    def world_to_plane(origin, normal):
        """Ma trận chuyển đổi từ World sang hệ toạ độ phẳng (World To Plane)"""
        M_p2w = Affine4x4.plane_to_world(origin, normal)
        return np.linalg.inv(M_p2w)

    @staticmethod
    def aligned_system(target_dir, hint_up=np.array([0.0, 1.0, 0.0])):
        """Aligned Coordinate System (Căn chỉnh trục cục bộ theo vector chỉ định)"""
        Z = np.array(target_dir, dtype=np.float64)
        Z = Z / np.linalg.norm(Z)
        
        X = np.cross(hint_up, Z)
        if np.linalg.norm(X) < 1e-6:
            X = np.cross(np.array([1.0, 0.0, 0.0]), Z)
        X = X / np.linalg.norm(X)
        Y = np.cross(Z, X)
        
        M = np.eye(4)
        M[:3, 0] = X
        M[:3, 1] = Y
        M[:3, 2] = Z
        return M

    @staticmethod
    def invert(matrix):
        """Nghịch đảo ma trận 4x4 M^-1"""
        return np.linalg.inv(matrix)
